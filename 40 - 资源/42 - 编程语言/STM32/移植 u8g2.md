本文档记录了在基于 **STM32F407VGT6** 标准库的现代化开发环境下，如何高效、轻量地移植 **U8g2** 图形库。重点解决了 U8g2 库体积过大导致的编译卡顿、RAM 溢出以及文本显示失效等常见问题。
## 开发环境准备
### 环境参数
- **MCU 型号**：STM32F407VGT6
- **屏幕信息**：0.96 寸 OLED (SSD1306, 128\*64, IIC 接口)
- **引脚连接**：SCL -> **PB15**, SDA -> **PD10**
- **软件组合**：Keil uVision 5 + VSCode (Clangd)
---
### 目录结构
由于 U8g2 属于大型第三方库，推荐在工程根目录新建 **`LIB`** 文件夹以存放第三方库
1. **LIB/U8G2**：存放精简后的 U8g2 核心源码
2. **HARDWARE/src**：存放针对当前硬件引脚编写的适配代码
> [!tip]
> 本文档没有采用标准库模板中**HARDWARE**下每个模块单独一共文件的操作而是直接将用户自定义的硬件驱动的源代码放在 `src` 下头文件放在 `inc` 下且默认 `inc` 已经添加到 keil 头文件的检测路径中

## U8g2 库的移植
### 库文件提取

U8g2 源码包含数多个驱动文件手动筛选容易出现问题，使用 Python 脚本在 `TOOLS` 目录下自动化提取 SSD1306 所需的最小文件集，避免手动筛选

- 在 TOOLS 下创建 `u8g2_copy.py`
``` python
import os, shutil, glob

# 1. 配置源码路径与目标路径
SOURCE_DIR = r"D:\Downloads\u8g2-2.29.11\csrc" # 修改为你下载的源码路径
TARGET_DIR = os.path.join("..", "LIB", "U8G2")
SPECIFIC_DRIVER = "u8x8_d_ssd1306_128x64_noname.c"

def clean_and_copy():
    abs_target = os.path.abspath(TARGET_DIR)
    if os.path.exists(abs_target): shutil.rmtree(abs_target)
    os.makedirs(abs_target, exist_ok=True)

    # 提取头文件、核心算法代码、协议逻辑代码
    files = glob.glob(os.path.join(SOURCE_DIR, "*.h")) + \
            glob.glob(os.path.join(SOURCE_DIR, "u8g2_*.c")) + \
            glob.glob(os.path.join(SOURCE_DIR, "u8x8_*.c")) + \
            glob.glob(os.path.join(SOURCE_DIR, "mui*.c")) + \
            glob.glob(os.path.join(SOURCE_DIR, "u8log*.c"))
            
    # 过滤掉非 SSD1306 的驱动文件，保持目录相对清爽
    files = [f for f in files if not os.path.basename(f).startswith("u8x8_d_") or SPECIFIC_DRIVER in f]

    for f in files: shutil.copy2(f, abs_target)
    print(f"Successfully copied {len(files)} files to {TARGET_DIR}")

if __name__ == "__main__": clean_and_copy()
```

### 驱动源码裁剪
为解决 RAM 溢出与编译卡顿，执行以下“手术”：

打开 `u8g2_d_setup.c`，仅保留 `u8g2_Setup_ssd1306_i2c_128x64_noname_1` 函数，删除其余数百个无效驱动
参考文件：
```c
/* u8g2_d_setup.c */
/* generated code, codebuild, u8g2 project */

#include "u8g2.h"
void u8g2_Setup_ssd1306_i2c_128x64_noname_1(u8g2_t *u8g2, const u8g2_cb_t *rotation, u8x8_msg_cb byte_cb, u8x8_msg_cb gpio_and_delay_cb)
{
  uint8_t tile_buf_height;
  uint8_t *buf;
  u8g2_SetupDisplay(u8g2, u8x8_d_ssd1306_128x64_noname, u8x8_cad_ssd13xx_fast_i2c, byte_cb, gpio_and_delay_cb);
  buf = u8g2_m_16_8_1(&tile_buf_height);
  u8g2_SetupBuffer(u8g2, buf, tile_buf_height, u8g2_ll_hvline_vertical_top_lsb, rotation);
}
```
### 自定义极简字体库

- 原始 `u8g2_fonts.c` 文件体积高达 **30 MB**，包含上千种字体，直接加载会导致 Keil 编译和索引极度缓慢为此新建新建 `u8g2_fonts_mini.c`，仅从原库中拷贝需要的字体数解决此问题

 [u8g2_fonts_mini.c](../../49%20-%20代码片段/u8g2_fonts_mini.c) 仅保留项目中确切需要的 3 种字体数据，以确保工程的轻量化。

```c
#include "u8g2.h"

// 1. 复古等宽数字字体 (仅 0-9 数字及少量符号，极致节省)
const uint8_t u8g2_font_amstrad_cpc_extended_8n[239] = 
    "\22\3\4\3\4\4\1\1\5\10\7\0\0\10\0\10\1\0\0\0\0\0\322 "
    /* ...省略中间数据... */
    "\343\7\22\23\27\223\5\0\0\0\4\377\377\0";

// 2. 5x7 像素极小字体 (全 ASCII 字符，高密度显示首选)
const uint8_t u8g2_font_5x7_tf[1677] = 
    "\277\0\2\2\3\3\3\4\4\5\7\0\377\6\377\6\0\1\22\2/\6p \5\0\275\1!\6\261\261"
    /* ...省略中间数据... */
    "\1\0\0\0\4\377\377\0";

// 3. 8x13 像素标准字体 (全 ASCII 字符，主界面显示首选)
const uint8_t u8g2_font_8x13_tf[2395] = 
    "\277\0\3\2\3\4\3\5\5\10\15\0\376\11\376\11\0\1r\2\363\11> \5\0n\14!\7\311C"
    /* ...省略中间数据... */
    "\1\0\0\0\4\377\377\0";
```

为了让 `main.c` 能够跨文件访问这些字体，需要建立配套的 `extern` 声明头文件 [u8g2_fonts_mini.h](../../49%20-%20代码片段/u8g2_fonts_mini.h)

```C
#ifndef __U8G2_FONTS_MINI_H
#define __U8G2_FONTS_MINI_H

#include <stdint.h>

/* 字体声明：确保变量名与 .c 文件中完全一致 */
extern const uint8_t u8g2_font_amstrad_cpc_extended_8n[];
extern const uint8_t u8g2_font_5x7_tf[];
extern const uint8_t u8g2_font_8x13_tf[];

#endif
```

> [!tip] 使用提示
> - 需要把源文件添加到工程中去再在头文件中引用才可使用
> - 若需显示中文字符，需在 `u8g2_fonts_mini.c` 中添加对应的中文字模数组，并在 `u8g2_fonts_mini.h` 中同步声明

## 硬件接口适配
在 `HARDWARE` 下实现回调函数，将 U8g2 的逻辑指令映射到 STM32 的 GPIO 操作上。
`HARDWARE/inc/u8g2_port.h`：

```c
#ifndef __U8G2_PORT_H
#define __U8G2_PORT_H

#include "stm32f4xx.h"
#include "u8g2.h"

#define OLED_SCL_PIN GPIO_Pin_15
#define OLED_SCL_PORT GPIOB
#define OLED_SDA_PIN GPIO_Pin_10
#define OLED_SDA_PORT GPIOD

// 初始化函数
void OLED_Init(u8g2_t *u8g2);

#endif
```

`HARDWARE/src/u8g2_port.c` ：

```c
#include "u8g2_port.h"
#include "delay.h"

/* 定义静态数组作为实际存储空间 */
/* 如果你用的是 _1 (单页模式)，大小是 128 */
static uint8_t oled_page_buffer[128]; 

/* 实现 U8g2 期待的函数接口 */
uint8_t *u8g2_m_16_8_1(uint8_t *page_cnt) {
    *page_cnt = 1;             // 告诉 U8g2 这是 1 页缓冲
    return oled_page_buffer;   // 返回缓冲区首地址
}

// U8g2 指令回调函数
uint8_t u8x8_gpio_and_delay_stm32(u8x8_t *u8x8, uint8_t msg, uint8_t arg_int,
                                  void *arg_ptr) {
    switch (msg) {
    case U8X8_MSG_DELAY_MILLI:
        delay_ms(arg_int);
        break;

    case U8X8_MSG_GPIO_AND_DELAY_INIT:
        // 同时使能 GPIOB 和 GPIOD 的硬件时钟
        RCC_AHB1PeriphClockCmd(RCC_AHB1Periph_GPIOB | RCC_AHB1Periph_GPIOD,
                               ENABLE);

        GPIO_InitTypeDef GPIO_InitStructure;
        GPIO_InitStructure.GPIO_Mode = GPIO_Mode_OUT;
        GPIO_InitStructure.GPIO_OType = GPIO_OType_OD; // 开漏输出
        GPIO_InitStructure.GPIO_Speed = GPIO_High_Speed;
        GPIO_InitStructure.GPIO_PuPd = GPIO_PuPd_NOPULL;

        // 初始化 PB15 (SCL)
        GPIO_InitStructure.GPIO_Pin = OLED_SCL_PIN;
        GPIO_Init(OLED_SCL_PORT, &GPIO_InitStructure);

        // 初始化 PD10 (SDA)
        GPIO_InitStructure.GPIO_Pin = OLED_SDA_PIN;
        GPIO_Init(OLED_SDA_PORT, &GPIO_InitStructure);
        break;

    case U8X8_MSG_GPIO_I2C_CLOCK:
        // 控制 SCL (PB15)
        GPIO_WriteBit(OLED_SCL_PORT, OLED_SCL_PIN, (BitAction)arg_int);
        break;

    case U8X8_MSG_GPIO_I2C_DATA:
        // 控制 SDA (PD10)
        GPIO_WriteBit(OLED_SDA_PORT, OLED_SDA_PIN, (BitAction)arg_int);
        break;

    default:
        return 0;
    }
    return 1;
}

void OLED_Init(u8g2_t *u8g2) {
    // SSD1306 128x64 软件 I2C 初始化
    u8g2_Setup_ssd1306_i2c_128x64_noname_1(u8g2, U8G2_R0, u8x8_byte_sw_i2c,
                                           u8x8_gpio_and_delay_stm32);
    u8g2_InitDisplay(u8g2);
    u8g2_SetPowerSave(u8g2, 0);
    u8g2_ClearBuffer(u8g2);
}

```

## 开发集成与环境优化
### Keil MDK 工程配置
1. **添加 Group**：
    - `LIB/U8G2`：添加 `LIB/U8G2` 目录下的全部 `.c` 文件
    - 排除 `mui.c`、`mui_u8g2.c`、`u8g2_d_memory.c`、`u8x8_fonts.c`
    - `HARDWARE`：添加 `u8g2_port.c`
2. **头文件路径**：
    - **Include Paths**：添加 `.\LIB\U8G2`
    - 魔法棒-> `C/C++` -> `Include Paths` -> 添加 `.\LIB\U8G2`
3. **消除警告**：
    - 所有的 `.c` 文件末尾必须保留一个**空行**，否则会触发 `#1-D: last line ends without a newline` 警告
---

### VSCode 跨工具链环境同步
每次在 Keil 中新增文件或修改路径后，必须运行 Python 脚本更新编译数据库：
1. **终端执行**：
    ```python
    python TOOLS/keil2json.py
    ```
2. **重启服务**：`F1` -> `Clangd: Restart language server`。

## 示例代码
该示例演示了三类字体的混排显示、标题分割线、几何进度条及圆形运行指示器
```c
#include "stm32f4xx.h"
#include "delay.h"
#include "u8g2_port.h"
#include "u8g2_fonts_mini.h"
#include <stdio.h>
u8g2_t u8g2;

int main(void) {
    uint8_t count = 0;
    char count_str[5];

    // 1. 系统基础初始化
    DELAY_SysTickInit(168); 
    
    // 2. OLED 硬件与 U8g2 协议初始化
    OLED_Init(&u8g2);

    while (1) {
        // --- U8g2 单页刷新循环开始 ---
        u8g2_FirstPage(&u8g2);
        do {
            // A. 绘制标题（使用标准 8x13 字体）
            u8g2_SetFont(&u8g2, u8g2_font_8x13_tf);
            u8g2_DrawStr(&u8g2, 0, 13, "STM32F407 VGT6");
            u8g2_DrawHLine(&u8g2, 0, 16, 128); // 标题分割线

            // B. 绘制复古动态数值（使用 Amstrad 8n 数字字体）
            u8g2_SetFont(&u8g2, u8g2_font_amstrad_cpc_extended_8n);
            sprintf(count_str, "%d", count);
            u8g2_DrawStr(&u8g2, 0, 42, "VAL:");
            u8g2_DrawStr(&u8g2, 45, 42, count_str);

            // C. 绘制进度条（几何图形演示）
            u8g2_DrawFrame(&u8g2, 0, 48, 128, 10);      // 进度条外框
            u8g2_DrawBox(&u8g2, 2, 50, count % 125, 6); // 内部填充块

            // D. 绘制页脚状态（使用 5x7 极小字体）
            u8g2_SetFont(&u8g2, u8g2_font_5x7_tf);
            u8g2_DrawStr(&u8g2, 0, 63, "System: Running...");
            u8g2_DrawCircle(&u8g2, 120, 58, 3, U8G2_DRAW_ALL); // 右下角运行指示点

        } while (u8g2_NextPage(&u8g2));
        // --- U8g2 刷新循环结束 ---

        // 变量逻辑处理
        count++;
        if (count > 124) count = 0;
        
        delay_ms(10); // 控制动画刷新速率
    }
}

```
