## 问题的提出
在网络服务器开发中，无论是 UDP 还是 TCP，核心挑战在于如何高效处理多个客户端的并发请求。
- **UDP 的挑战**：默认的 `recvfrom` 是阻塞的。如果服务端调用它但客户端没发数据，服务端就会永久阻塞，无法处理其他事务。
- **TCP 的挑战**：TCP 是面向连接的。每当有一个客户端连接成功，服务端都会产生一个新的已连接套接字。随着连接数增加，服务端必须妥善管理这些套接字。
**IO 模型**就是指当服务端需要同时处理多个套接字时采取的策略。
- 常见的模型
	- 非阻塞轮询
	- 多任务并发
	- 异步信号驱动
	- 多路复用
## 非阻塞轮询
### 核心思想
将套接字设置为**非阻塞 (Non-blocking)** 状态。在此状态下，如果读取操作无法立即获取数据，API 不会阻塞，而是返回错误（通常是 `EAGAIN`）。服务端需要在一个循环中不断尝试读取（即“轮询”）
### 优缺点
- **优点**：避免了进程死等造成的假死。
- **缺点**：CPU 占用极高（通常单核 100%），因为大部分时间都在做无用的空转查询。**不推荐用于生产环境**。
### 2.3 API 设置
使用 `fcntl` 修改文件描述符属性。
```c
// 设置为非阻塞
long state = fcntl(sockfd, F_GETFL); // 1. 获取原有属性
state |= O_NONBLOCK;                 // 2. 加上非阻塞位
fcntl(sockfd, F_SETFL, state);       // 3. 写回属性

// 恢复为阻塞
long state = fcntl(sockfd, F_GETFL);
state &= ~O_NONBLOCK;                // 去掉非阻塞位
fcntl(sockfd, F_SETFL, state);
```
### 代码示例
```c
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <string.h>
#include <strings.h>
#include <errno.h>
#include <fcntl.h>
#include <sys/socket.h>
#include <arpa/inet.h>

#define PORT 50023

int main(int argc, char const *argv[])
{
    // 1. 创建UDP套接字
    int sockfd = socket(AF_INET, SOCK_DGRAM, IPPROTO_UDP);
    if (sockfd < 0) { perror("socket"); exit(1); }

    struct sockaddr_in addr = {0};
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = htonl(INADDR_ANY); 
    addr.sin_port = htons(PORT); 

    // 2. 绑定地址
    if (bind(sockfd, (struct sockaddr *)&addr, sizeof(addr)) < 0) {
        perror("bind"); exit(1);
    }
    printf("UDP服务器启动，正在轮询端口 %d...\n", PORT);

    // 3. 将套接字设定为非阻塞
    long state = fcntl(sockfd, F_GETFL);
    state |= O_NONBLOCK;
    fcntl(sockfd, F_SETFL, state);

    char buf[100];
    struct sockaddr_in clientAddr;
    socklen_t len;

    // 4. 轮询接收
    while(1)
    {
        bzero(buf, 100);
        len = sizeof(clientAddr);
        bzero(&clientAddr, len);

        // 尝试读取
        int n = recvfrom(sockfd, buf, 100, 0, (struct sockaddr *)&clientAddr, &len);

        if(n > 0) {
            // 正常读到数据
            printf("收到 [%s:%d]: %s\n", 
                   inet_ntoa(clientAddr.sin_addr), ntohs(clientAddr.sin_port), buf);
        }
        else if(n == -1 && errno == EAGAIN) {
            // 资源暂时不可得，继续轮询
            // 可以在这里做一点别的事情，或者稍微休眠一下降低CPU消耗
            // usleep(1000); 
            continue;
        }
        else {
            perror("recvfrom error");
        }
    }

    close(sockfd);
    return 0;
}
```

## 多任务并发
### 3.1 核心思想

利用多进程或多线程，为每一个客户端连接分配一个独立的执行单元。
- **多线程**：资源开销小，通信方便，适合逻辑处理较为简单的 TCP 通信
- **多进程**：隔离性好，但开销大，适合逻辑复杂的业务
### 设计模式
1. **主线程**：负责 `accept()` 监听套接字，等待连接。
2. **工作线程**：每当 `accept` 返回一个新的 `connfd`，就创建一个新线程专门负责与该客户端通信。
#### 代码示例
```c
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <string.h>
#include <pthread.h>
#include <sys/socket.h>
#include <arpa/inet.h>

#define PORT 50024

// 定义传递给线程的参数结构体
struct client_info {
    int connfd;
    struct sockaddr_in addr;
};

// 客户端处理线程函数
void *clientRoutine(void *arg)
{
    struct client_info *pInfo = (struct client_info *)arg;
    int connfd = pInfo->connfd;
    char buf[1024];

    // 线程分离，结束时自动回收资源
    pthread_detach(pthread_self());

    printf("线程 [ID:%lu] 开始服务客户端 %s:%d\n", 
           pthread_self(), inet_ntoa(pInfo->addr.sin_addr), ntohs(pInfo->addr.sin_port));

    while(1) {
        bzero(buf, sizeof(buf));
        int n = read(connfd, buf, sizeof(buf));
        
        if(n <= 0) {
            printf("客户端 %s:%d 断开连接\n", 
                   inet_ntoa(pInfo->addr.sin_addr), ntohs(pInfo->addr.sin_port));
            break;
        }
        printf("收到消息: %s", buf);
        
        // 回显
        write(connfd, buf, n);
    }

    close(connfd);
    free(pInfo); // 释放主线程分配的内存
    return NULL;
}

int main(int argc, char **argv)
{
    // 1. 准备TCP监听套接字
    int sockfd = socket(AF_INET, SOCK_STREAM, 0);
    if (sockfd < 0) { perror("socket"); exit(1); }

    // 端口复用设置（避免 bind 失败）
    int on = 1;
    setsockopt(sockfd, SOL_SOCKET, SO_REUSEADDR, &on, sizeof(on));

    struct sockaddr_in addr = {0};
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = htonl(INADDR_ANY);
    addr.sin_port = htons(PORT);

    if (bind(sockfd, (struct sockaddr *)&addr, sizeof(addr)) < 0) {
        perror("bind"); exit(1);
    }

    if (listen(sockfd, 5) < 0) { perror("listen"); exit(1); }
    printf("TCP 多线程服务器启动，监听端口 %d...\n", PORT);

    // 2. 主循环：接受连接并派发线程
    while(1)
    {
        struct client_info *newClient = malloc(sizeof(struct client_info));
        socklen_t len = sizeof(newClient->addr);

        // 阻塞等待连接
        if((newClient->connfd = accept(sockfd, (struct sockaddr *)&newClient->addr, &len)) < 0)
        {
            perror("accept失败");
            free(newClient);
            continue;
        }

        // 3. 创建专用线程处理该客户端
        pthread_t tid;
        if (pthread_create(&tid, NULL, clientRoutine, (void *)newClient) != 0) {
            perror("线程创建失败");
            close(newClient->connfd);
            free(newClient);
        }
    }
    return 0;
}
```

### 异步信号驱动
#### 核心思想
利用内核信号机制。当套接字有数据到达时，内核发送 `SIGIO` (信号 29) 给应用进程，应用进程在信号处理函数中读取数据。
#### 适用性
- **适合 UDP**：UDP 无连接，只有数据到达这一种事件。
- **不适合 TCP**：TCP 的连接建立、数据到达、断开请求等都会触发同一个 `SIGIO`，难以区分，处理复杂。
#### 设置步骤
1. **注册信号**：`signal(SIGIO, handler)`。
2. **设置属主**：`fcntl(sockfd, F_SETOWN, getpid())`，告诉内核信号发给谁。
3. **开启异步**：设置 `O_ASYNC` 标志。
#### 完整代码示例 (UDP 信号驱动)
```c
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <string.h>
#include <fcntl.h>
#include <signal.h>
#include <sys/socket.h>
#include <arpa/inet.h>

#define PORT 50025

int sockfd; // 全局变量，以便信号处理函数访问

// 信号响应函数
void f(int sig)
{
    if (sig == SIGIO) {
        struct sockaddr_in clientAddr;
        char buf[100];
        socklen_t len = sizeof(clientAddr);
        
        bzero(buf, 100);
        // 在信号处理函数中接收数据
        int n = recvfrom(sockfd, buf, 100, 0, (struct sockaddr *)&clientAddr, &len);
        if (n > 0) {
            printf("\n[SIGIO触发] 收到来自 [%s:%hu]: %s", 
                   inet_ntoa(clientAddr.sin_addr), ntohs(clientAddr.sin_port), buf);
        }
    }
}

int main(int argc, char const *argv[])
{
    // 1. 创建并绑定UDP套接字
    sockfd = socket(AF_INET, SOCK_DGRAM, 0);
    struct sockaddr_in addr = {0};
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = htonl(INADDR_ANY);
    addr.sin_port = htons(PORT);
    
    if (bind(sockfd, (struct sockaddr *)&addr, sizeof(addr)) < 0) {
        perror("bind"); exit(1);
    }
    printf("UDP 信号驱动服务器启动 (PID: %d)...\n", getpid());

    // 2. 步骤一：捕捉信号 SIGIO
    signal(SIGIO, f);

    // 3. 步骤二：设置信号的属主 (当前进程)
    fcntl(sockfd, F_SETOWN, getpid());

    // 4. 步骤三：将套接字设置为异步工作模式
    long state = fcntl(sockfd, F_GETFL);
    state |= O_ASYNC;
    fcntl(sockfd, F_SETFL, state);

    // 5. 主程序可以做其他事情 (模拟繁忙的主线程)
    printf("主线程正在执行其他任务(打印字母)...\n");
    for(int i=0; ; i++, i%=26)
    {
        fprintf(stderr, "%c", 'a'+i); // 打印 a-z
        usleep(200 * 1000); // 200ms
    }

    return 0;
}
```
### 多路复用
#### 核心思想
通过 select 或 poll 系统调用，由内核代替应用程序同时监控多个套接字的读、写、异常状态。一旦有一个或多个套接字就绪，函数返回，应用程序再进行处理。
这种方式既不需要多进程/线程，也不需要轮询，非常高效。
#### 就绪状态定义
- **读就绪**：有数据可读 (`recv`)，或有新连接请求 (`accept`)。
- **写就绪**：发送缓冲区有空间 (`send`)。
- **异常就绪**：收到 TCP 带外数据 (OOB)。
#### Select 实现
`select` 使用 `fd_set` 集合来管理文件描述符。
**接口：**
```c
int select(int nfds, fd_set *readfds, fd_set *writefds, fd_set *exceptfds, struct timeval *timeout);
```
**注意：**
1. `nfds` 必须是所有被监控 fd 中最大值 + 1。
2. `select` 返回后，集合中未就绪的位会被清除，因此**每次循环都要重新 `FD_SET`**。
```c
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <string.h>
#include <sys/socket.h>
#include <arpa/inet.h>
#include <sys/select.h>

#define PORT 50026
#define MAX_CLIENTS 10

int main(int argc, char **argv)
{
    // 1. 准备监听套接字
    int listenfd = socket(AF_INET, SOCK_STREAM, 0);
    struct sockaddr_in addr = {0};
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = htonl(INADDR_ANY);
    addr.sin_port = htons(PORT);
    
    // 设置地址复用
    int on = 1;
    setsockopt(listenfd, SOL_SOCKET, SO_REUSEADDR, &on, sizeof(on));

    bind(listenfd, (struct sockaddr *)&addr, sizeof(addr));
    listen(listenfd, 5);

    printf("TCP Select服务器启动，端口 %d\n", PORT);

    // 客户端套接字数组
    int client_fds[MAX_CLIENTS];
    for(int i=0; i<MAX_CLIENTS; i++) client_fds[i] = 0;

    fd_set rset;
    int maxfd;

    while(1)
    {
        // 2. 初始化集合
        FD_ZERO(&rset);
        FD_SET(listenfd, &rset); // 加入监听套接字
        maxfd = listenfd;

        // 将已连接的客户端加入集合
        for(int i=0; i<MAX_CLIENTS; i++) {
            int fd = client_fds[i];
            if(fd > 0) {
                FD_SET(fd, &rset);
            }
            if(fd > maxfd) maxfd = fd;
        }

        // 3. 阻塞等待就绪
        int activity = select(maxfd + 1, &rset, NULL, NULL, NULL);

        if (activity < 0) {
            perror("select error"); continue;
        }

        // 4. 检查监听套接字 (处理新连接)
        if (FD_ISSET(listenfd, &rset)) {
            struct sockaddr_in cli_addr;
            socklen_t len = sizeof(cli_addr);
            int new_sock = accept(listenfd, (struct sockaddr *)&cli_addr, &len);
            
            printf("新连接: socket %d, ip %s\n", new_sock, inet_ntoa(cli_addr.sin_addr));

            // 加入数组
            for(int i=0; i<MAX_CLIENTS; i++) {
                if(client_fds[i] == 0) {
                    client_fds[i] = new_sock;
                    break;
                }
            }
        }

        // 5. 检查所有客户端套接字 (处理数据)
        for(int i=0; i<MAX_CLIENTS; i++) {
            int fd = client_fds[i];
            if (fd > 0 && FD_ISSET(fd, &rset)) {
                char buf[1024] = {0};
                int n = read(fd, buf, 1024);
                
                if (n == 0) {
                    // 客户端断开
                    printf("客户端 socket %d 断开连接\n", fd);
                    close(fd);
                    client_fds[i] = 0;
                } else {
                    printf("收到来自 %d 的数据: %s", fd, buf);
                    write(fd, buf, n); // 回显
                }
            }
        }
    }
    return 0;
}
```

#### Poll 实现
`poll` 使用 `pollfd` 结构体数组，突破了 `select` 的最大文件描述符限制（通常 1024），且接口使用更统一。
**接口：**
```c
int poll(struct pollfd *fds, nfds_t nfds, int timeout);
struct pollfd {
    int   fd;         // 文件描述符
    short events;     // 监控的事件 (输入)
    short revents;    // 发生的事件 (输出)
};
```
- 事件宏：`POLLIN` (读), `POLLOUT` (写), `POLLPRI` (异常/OOB)。
**完整代码示例 (TCP Poll 服务器)：**
```c
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <string.h>
#include <sys/socket.h>
#include <arpa/inet.h>
#include <poll.h>

#define PORT 50027
#define MAX_OPEN_FDS 100

int main(int argc, char **argv)
{
    int listenfd = socket(AF_INET, SOCK_STREAM, 0);
    // ... bind, listen 省略错误检查 ...
    struct sockaddr_in addr = {0};
    addr.sin_family = AF_INET; addr.sin_addr.s_addr = htonl(INADDR_ANY); addr.sin_port = htons(PORT);
    int on=1; setsockopt(listenfd, SOL_SOCKET, SO_REUSEADDR, &on, sizeof(on));
    bind(listenfd, (struct sockaddr *)&addr, sizeof(addr));
    listen(listenfd, 5);

    printf("TCP Poll服务器启动，端口 %d\n", PORT);

    struct pollfd fds[MAX_OPEN_FDS];
    // 初始化：fds[0] 为监听套接字
    fds[0].fd = listenfd;
    fds[0].events = POLLIN; // 监控读就绪
    
    int nfds = 1; // 当前监控的套接字数量

    while(1) {
        // 1. 调用 poll, timeout = -1 (无限等待)
        int n = poll(fds, nfds, -1);
        if(n < 0) { perror("poll"); break; }

        int current_nfds = nfds; // 遍历时防止 nfds 变化影响循环
        for(int i=0; i < current_nfds; i++) {
            
            // 检查是否就绪
            if(fds[i].revents == 0) continue;

            // 2. 处理监听套接字：新连接
            if(fds[i].fd == listenfd && (fds[i].revents & POLLIN)) {
                struct sockaddr_in cli_addr;
                socklen_t len = sizeof(cli_addr);
                int new_sock = accept(listenfd, (struct sockaddr *)&cli_addr, &len);
                printf("新连接: %d\n", new_sock);

                if(nfds < MAX_OPEN_FDS) {
                    fds[nfds].fd = new_sock;
                    fds[nfds].events = POLLIN | POLLPRI; // 监控普通读和异常(OOB)
                    fds[nfds].revents = 0;
                    nfds++;
                } else {
                    printf("连接已满\n");
                    close(new_sock);
                }
            }
            // 3. 处理客户端套接字：普通数据
            else if(fds[i].revents & POLLIN) {
                char buf[1024] = {0};
                int n_read = read(fds[i].fd, buf, sizeof(buf));
                if(n_read <= 0) {
                    close(fds[i].fd);
                    // 简单的删除操作：将最后一个移到当前位置
                    fds[i] = fds[nfds-1];
                    nfds--;
                    i--; // 重新检查当前位置
                } else {
                    printf("收到数据: %s", buf);
                }
            }
            // 4. 处理带外数据 (OOB)
            else if(fds[i].revents & POLLPRI) {
                char oob_byte;
                recv(fds[i].fd, &oob_byte, 1, MSG_OOB);
                printf("收到 OOB 数据: %c\n", oob_byte);
            }
        }
    }
    return 0;
}
```

### 6. 总结：什么时候用哪种？

|**模型**|**适用场景**|**备注**|
|---|---|---|
|**阻塞与轮询**|不推荐|浪费 CPU 资源|
|**多任务并发**|TCP 服务端|逻辑清晰，编码简单，最常见|
|**信号驱动**|UDP 服务端|避免轮询，简单易用|
|**多路复用**|高并发/资源受限|单线程处理大量并发，无需创建大量线程，高效|
